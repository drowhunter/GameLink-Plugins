    public class ProfileManager : ObservableObject
    {
        private float[] outputArray = new float[6], prevOutputArray = new float[6];
        private float[] prevValues = Array.Empty<float>(), prevRawValues = Array.Empty<float>();
        private SpikeData[] spikeData = Array.Empty<SpikeData>();
        private float previousYaw, yawOffset;
        private bool resetyawFlag;
        private YawController controller = new();
        private Smooth m_smoothPitch = new(), m_smoothRoll = new();
        public float[] InputArray = new float[15], ComponentOutput = Array.Empty<float>();
        public PluginProfiles Profile = new();
        public bool OverrideMovement;
        public Vector3 overridePos = new();
        public float GlobalWashout = 1;
        public void Configure(ObservableCollection<Profile_Component> components, bool antijump,
            int threshold, float washout)
        {
            CoreContext.Instance = new();
            Settings.Instance = new() { Antijump = antijump, AntijumpValue = threshold };
            Profile.ActiveProfile.Components = components;
            GlobalWashout = washout;
        }
        public void RequestYawReset() => resetyawFlag = true;
        public float GetCapturedOutput(int index) => index switch
        {
            0 => (float)controller.Yaw,
            1 => (float)controller.Pitch,
            2 => (float)controller.Roll,
            _ => outputArray[index]
        };
        public float Step(float[] inputs)
        {
            InputArray = inputs;
            CalculateOutput();
            return (float)controller.Yaw;
        }
        /* PROFILE_METHODS */
    }
